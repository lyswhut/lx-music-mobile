cmake_minimum_required(VERSION 3.16)
project(lx_user_api_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(../../quickjs quickjs)
add_executable(script_runtime_test ScriptRuntimeTest.cpp ../ScriptRuntime.cpp)
target_include_directories(script_runtime_test PRIVATE ..)
target_link_libraries(script_runtime_test PRIVATE lx_quickjs)
set(LX_PRELOAD ${CMAKE_CURRENT_SOURCE_DIR}/../../../resources/rawfile/user-api-preload.js)
add_test(NAME script_runtime COMMAND script_runtime_test ${LX_PRELOAD})
set_tests_properties(script_runtime PROPERTIES TIMEOUT 30)

if(LX_NODE_INCLUDE AND LX_NODE_EXECUTABLE)
  add_library(script_bridge_test SHARED ../NativeBridge.cpp ../ScriptRuntime.cpp)
  target_compile_definitions(script_bridge_test PRIVATE LX_USER_API_NODE_TEST)
  target_include_directories(script_bridge_test PRIVATE ${LX_NODE_INCLUDE})
  target_link_libraries(script_bridge_test PRIVATE lx_quickjs)
  set_target_properties(script_bridge_test PROPERTIES PREFIX "" SUFFIX ".node")
  add_test(NAME script_bridge COMMAND ${LX_NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/NativeBridgeTest.cjs $<TARGET_FILE:script_bridge_test> ${LX_PRELOAD})
  set_tests_properties(script_bridge PROPERTIES TIMEOUT 15)
  add_test(NAME worker_host COMMAND ${LX_NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/WorkerHostTest.cjs $<TARGET_FILE:script_bridge_test> ${LX_PRELOAD})
  set_tests_properties(worker_host PROPERTIES TIMEOUT 70)
endif()
